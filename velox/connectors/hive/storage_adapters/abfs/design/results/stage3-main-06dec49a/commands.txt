ASAN
# Configure as documented by the Stage 2 sanitizer environment with Clang 20,
# bundled Folly, and -fsanitize=address. After CMake regeneration, set the
# generated bundled-Folly library-ASAN marker to 1 and rebuild.
cmake --build abfs-stage2-asan-clang20-debug --target \
  velox_abfs_test velox_abfs_registration_test \
  velox_abfs_stage1_transport_test velox_abfs_stage2_runtime_test -j2
export ASAN_OPTIONS=detect_leaks=1:halt_on_error=1:abort_on_error=1
ctest --test-dir abfs-stage2-asan-clang20-debug \
  -R '^(velox_abfs_test|velox_abfs_registration_test|velox_abfs_stage1_transport_test|velox_abfs_stage2_runtime_test)$' \
  --output-on-failure
